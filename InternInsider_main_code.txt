
==================== backend/app/db.py ====================

"""
Database + embeddings. Everything that touches PostgreSQL or turns text into
a vector lives here.
"""
import hashlib
import math
import os
import re

import psycopg
from psycopg.rows import dict_row

DEFAULT_DATABASE_URL = "postgresql://postgres:postgres@localhost:5432/interninsider"
EMBEDDING_DIM = 384  # size of the vectors produced by the embedding model below

SCHEMA_SQL = f"""
CREATE EXTENSION IF NOT EXISTS vector;          -- turns on pgvector

CREATE TABLE IF NOT EXISTS jobs (
    id          SERIAL PRIMARY KEY,
    title       TEXT NOT NULL,
    company     TEXT NOT NULL,
    location    TEXT NOT NULL DEFAULT '',
    category    TEXT NOT NULL DEFAULT '',
    url         TEXT NOT NULL DEFAULT '',
    description TEXT NOT NULL,
    skills      TEXT[] NOT NULL DEFAULT '{{}}',   -- skills found in the description
    embedding   vector({EMBEDDING_DIM}) NOT NULL  -- the meaning of title + description as {EMBEDDING_DIM} numbers
);

-- HNSW is an approximate nearest-neighbour index. Without it Postgres compares
-- the query against every row (fine for hundreds of jobs, slow for millions).
CREATE INDEX IF NOT EXISTS idx_jobs_embedding
    ON jobs USING hnsw (embedding vector_cosine_ops);
"""


def connect():
    """Open a connection. Rows come back as dicts, e.g. row["title"]."""
    return psycopg.connect(
        os.environ.get("DATABASE_URL", DEFAULT_DATABASE_URL), row_factory=dict_row
    )


def init_db():
    with connect() as conn:
        conn.execute(SCHEMA_SQL)


# ------------------------------------------------------------- embeddings

_model = None


def embed(text: str) -> list[float]:
    """
    Turn text into a list of 384 numbers (an "embedding"). Texts with similar
    MEANING get vectors that point in similar directions, even when they share
    no words: "build user interfaces" lands near "React developer".

    The model (BAAI/bge-small-en-v1.5) runs locally through fastembed, so
    search needs no API key and costs nothing. The first call downloads it.
    """
    if os.environ.get("EMBEDDINGS") == "fake":
        return _fake_embed(text)
    global _model
    if _model is None:
        from fastembed import TextEmbedding

        _model = TextEmbedding("BAAI/bge-small-en-v1.5")
    return [float(x) for x in next(iter(_model.embed([text])))]


def _fake_embed(text: str) -> list[float]:
    """
    TEST ONLY (EMBEDDINGS=fake). No model, no download: each word is hashed
    into one of the 384 slots. It only matches shared words, not meaning, but
    it lets the tests exercise the real pgvector SQL quickly and offline.
    """
    vec = [0.0] * EMBEDDING_DIM
    for word in re.findall(r"[a-z0-9+#]+", text.lower()):
        slot = int(hashlib.md5(word.encode()).hexdigest(), 16) % EMBEDDING_DIM
        vec[slot] += 1.0
    length = math.sqrt(sum(x * x for x in vec)) or 1.0
    return [x / length for x in vec]


def to_pgvector(vec: list[float]) -> str:
    """pgvector accepts a vector written as text: '[0.1,0.2,...]'."""
    return "[" + ",".join(f"{x:.6f}" for x in vec) + "]"

==================== backend/app/skills.py ====================

"""
Skill extraction, the simple way: a fixed vocabulary and word matching.
Good enough for an MVP and easy to explain. Its weakness is that it only
finds skills that are in the list and spelled the way the list spells them.
"""
import re

SKILLS = [
    "python", "java", "javascript", "typescript", "go", "c++", "c#", "swift", "kotlin", "dart", "bash", "sql",
    "html", "css", "react", "react native", "vue", "angular", "next.js", "tailwind", "node.js", "express",
    "fastapi", "django", "flask", "spring boot", "graphql", "rest", "grpc",
    "postgresql", "mysql", "mongodb", "redis", "kafka", "rabbitmq",
    "docker", "kubernetes", "terraform", "aws", "azure", "gcp", "linux", "git", "jenkins", "github actions", "ci/cd",
    "pytorch", "tensorflow", "scikit-learn", "pandas", "numpy", "spark", "airflow", "dbt", "hugging face",
    "machine learning", "deep learning", "nlp", "llm", "embeddings", "statistics",
    "jest", "pytest", "unit testing", "figma", "firebase", "swiftui", "flutter", "jetpack compose",
    "networking", "penetration testing", "owasp", "siem", "prometheus", "grafana",
]

# Different spellings that mean the same skill.
ALIASES = {"postgres": "postgresql", "nodejs": "node.js", "k8s": "kubernetes",
           "sklearn": "scikit-learn", "golang": "go",
           "rest api": "rest", "rest apis": "rest", "unit tests": "unit testing", "huggingface": "hugging face"}


def extract_skills(text: str) -> list[str]:
    """Return the skills from SKILLS that appear in the text, alphabetically."""
    text = text.lower()
    found = set()
    for term in SKILLS + list(ALIASES):
        # (?<![a-z0-9]) and (?![a-z0-9]) mean "not in the middle of a word",
        # so "go" does not match inside "google" and "java" not inside "javascript".
        if re.search(r"(?<![a-z0-9])" + re.escape(term) + r"(?![a-z0-9+#])", text):
            found.add(ALIASES.get(term, term))
    return sorted(found)

==================== backend/app/tools.py ====================

"""
The 5 tools. Each one is a plain Python function with type hints and a
docstring. The SAME functions are used in three places:
  - the LangGraph agent   (agent.py)      the LLM decides when to call them
  - the MCP server        (mcp_server.py) other AI apps can call them
  - the REST API          (main.py)       the React page calls some directly
The docstrings matter: they are what the LLM reads to decide which tool to use.
"""
from app.db import connect, embed, to_pgvector
from app.skills import extract_skills


def search_jobs(query: str, limit: int = 5) -> list[dict]:
    """Search job postings by meaning. Use for requests like 'find backend internships'
    or 'jobs working with data'. Returns the closest postings with a similarity score (0 to 1)."""
    vector = to_pgvector(embed(query))
    with connect() as conn:
        # <=> is pgvector's cosine DISTANCE operator (0 = same direction).
        # Similarity is 1 - distance. ORDER BY distance = nearest first.
        rows = conn.execute(
            """
            SELECT id, title, company, location, skills,
                   round((1 - (embedding <=> %s::vector))::numeric, 3)::float AS similarity
            FROM jobs
            ORDER BY embedding <=> %s::vector
            LIMIT %s
            """,
            (vector, vector, limit),
        ).fetchall()
    return rows


def get_job(job_id: int) -> dict:
    """Get the full details of one job posting, including its description, by its id."""
    with connect() as conn:
        row = conn.execute(
            "SELECT id, title, company, location, category, url, description, skills FROM jobs WHERE id = %s",
            (job_id,),
        ).fetchone()
    return row or {"error": f"No job with id {job_id}"}


def match_resume(resume_text: str, limit: int = 5) -> list[dict]:
    """Find the job postings that best match a resume. Pass the full resume text.
    Each result also lists which of the job's skills the resume already has."""
    resume_skills = set(extract_skills(resume_text))
    jobs = search_jobs(resume_text, limit)  # a resume is just a long search query
    for job in jobs:
        job["matching_skills"] = sorted(resume_skills & set(job["skills"]))
    return jobs


def skill_gap(resume_text: str, job_id: int) -> dict:
    """Compare a resume against one job posting. Returns the skills the candidate
    already has, the skills that are missing, and a match percentage."""
    job = get_job(job_id)
    if "error" in job:
        return job
    job_skills = set(job["skills"])
    resume_skills = set(extract_skills(resume_text))
    have = sorted(job_skills & resume_skills)
    missing = sorted(job_skills - resume_skills)
    percent = round(100 * len(have) / len(job_skills)) if job_skills else 0
    return {"job_id": job_id, "title": job["title"], "company": job["company"],
            "have": have, "missing": missing, "match_percent": percent}


def interview_prep(job_id: int) -> dict:
    """Get the material needed to prepare for an interview for one job: the role,
    the skills to revise, and the posting text. Use this, then write practice questions from it."""
    job = get_job(job_id)
    if "error" in job:
        return job
    return {"job_id": job_id, "title": job["title"], "company": job["company"],
            "skills_to_revise": job["skills"], "posting": job["description"],
            "question_types": ["technical question per skill", "one project deep-dive",
                               "one behavioural question", "one question to ask the interviewer"]}


ALL_TOOLS = [search_jobs, get_job, match_resume, skill_gap, interview_prep]

==================== backend/app/agent.py ====================

"""
The LangGraph agent.

An "agent" is an LLM in a loop that can call tools. The graph has two nodes:

    START -> agent --(LLM asked for a tool)--> tools --+
               ^                                        |
               +----------------------------------------+
               |
               +--(LLM wrote a normal answer)--> END

  agent node: send the conversation to the LLM. It replies EITHER with a final
              answer OR with "please call tool X with these arguments".
  tools node: run the requested Python function and append its result to the
              conversation. Then go back to the agent node.
The loop ends when the LLM answers without asking for a tool.
"""
import os

from langchain.chat_models import init_chat_model
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from langgraph.graph import START, MessagesState, StateGraph
from langgraph.prebuilt import ToolNode, tools_condition

from app.tools import ALL_TOOLS

# "provider:model". Change it in backend/.env without touching the code.
DEFAULT_MODEL = "openai:gpt-5.4-mini"

SYSTEM_PROMPT = """You are InternInsider, a career copilot for students looking for internships.
You can search real job postings, match a resume to jobs, find skill gaps, and prepare interview questions.
Rules:
- Always use the tools to look up jobs. Never invent a job posting, company, or job id.
- When you mention a job, include its id, title and company.
- If a tool needs the resume and none was provided, ask the user to paste it.
- Keep answers short and practical. Write plain text, no markdown symbols."""


def build_agent(llm=None):
    """Build and compile the graph. Tests pass in a fake `llm`."""
    llm = llm or init_chat_model(os.environ.get("LLM_MODEL", DEFAULT_MODEL))
    # bind_tools sends each tool's name, docstring and argument types to the LLM.
    llm_with_tools = llm.bind_tools(ALL_TOOLS)

    def call_model(state: MessagesState):
        reply = llm_with_tools.invoke(state["messages"])
        return {"messages": [reply]}  # appended to the conversation

    graph = StateGraph(MessagesState)  # the state is just the list of messages
    graph.add_node("agent", call_model)
    graph.add_node("tools", ToolNode(ALL_TOOLS))
    graph.add_edge(START, "agent")
    # tools_condition looks at the LLM's last message: tool call -> "tools", else -> END
    graph.add_conditional_edges("agent", tools_condition)
    graph.add_edge("tools", "agent")
    return graph.compile()


def run_agent(agent, message: str, resume_text: str = "", history: list[dict] | None = None) -> dict:
    """Run one user turn. Returns the answer and which tools the agent called."""
    system = SYSTEM_PROMPT
    if resume_text.strip():
        system += "\n\nThe user's resume:\n" + resume_text.strip()

    messages = [SystemMessage(system)]
    for item in history or []:  # earlier turns, so follow-up questions make sense
        cls = HumanMessage if item["role"] == "user" else AIMessage
        messages.append(cls(item["content"]))
    messages.append(HumanMessage(message))

    # recursion_limit stops a confused LLM from looping on tools forever.
    result = agent.invoke({"messages": messages}, {"recursion_limit": 12})

    new_messages = result["messages"][len(messages):]
    tool_calls = [
        {"name": call["name"], "args": _shorten(call["args"])}
        for m in new_messages if isinstance(m, AIMessage)
        for call in m.tool_calls
    ]
    return {"reply": result["messages"][-1].text, "tool_calls": tool_calls}


def _shorten(args: dict) -> dict:
    """Trim long arguments (like a whole resume) so the UI can show them."""
    return {k: (v[:60] + "...") if isinstance(v, str) and len(v) > 60 else v for k, v in args.items()}

==================== backend/app/mcp_server.py ====================

"""
MCP server: the same 5 tools, offered to OTHER AI apps.

MCP (Model Context Protocol) is a standard way for an AI app to discover and
call tools. LangGraph is how OUR agent uses the tools. MCP is how SOMEONE
ELSE'S agent (Claude Desktop, an IDE assistant) can use them, without us
writing anything specific for that app.

Run it:   python -m app.mcp_server
It talks over stdin/stdout ("stdio"), so it prints nothing. The AI app starts
this process itself and exchanges JSON messages with it.
"""
from mcp.server import MCPServer

from app.tools import ALL_TOOLS

server = MCPServer("InternInsider")

for function in ALL_TOOLS:
    # Registers the function. Its name, docstring and type hints become the
    # tool description that the other AI app sees.
    server.tool()(function)

if __name__ == "__main__":
    server.run()  # stdio transport by default

==================== backend/app/main.py ====================

"""
FastAPI server: the HTTP front door for the React page.

Search, match and skill-gap endpoints call the tools directly, so they work
with NO API key. Only /api/chat needs the LLM.
"""
import os
from contextlib import asynccontextmanager

from dotenv import load_dotenv
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field

load_dotenv()  # reads backend/.env (OPENAI_API_KEY, LLM_MODEL, DATABASE_URL)

from app import tools  # noqa: E402
from app.agent import build_agent, run_agent  # noqa: E402
from app.db import connect, init_db  # noqa: E402
from app.skills import extract_skills  # noqa: E402


@asynccontextmanager
async def lifespan(app: FastAPI):
    init_db()  # create the table on startup if it is missing
    yield


app = FastAPI(title="InternInsider", lifespan=lifespan)

# DEMO ONLY: allow any website to call this API. In production, list the real frontend URL.
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

_agent = None  # built on the first chat request, then reused


class ResumeRequest(BaseModel):
    resume_text: str = Field(min_length=1, max_length=20000)


class SkillGapRequest(ResumeRequest):
    job_id: int


class TextRequest(BaseModel):
    text: str = Field(max_length=20000)


class ChatRequest(BaseModel):
    message: str = Field(min_length=1, max_length=4000)
    resume_text: str = Field(default="", max_length=20000)
    history: list[dict] = []  # [{"role": "user" | "assistant", "content": "..."}]


def llm_configured() -> bool:
    return bool(os.environ.get("OPENAI_API_KEY") or os.environ.get("LLM_MODEL"))


@app.get("/api/health")
def health():
    with connect() as conn:
        count = conn.execute("SELECT count(*) AS n FROM jobs").fetchone()["n"]
    return {"jobs": count, "llm_configured": llm_configured()}


@app.get("/api/jobs")
def list_jobs():
    """All postings, for the Jobs page before the user has searched."""
    with connect() as conn:
        return conn.execute(
            "SELECT id, title, company, location, category, skills FROM jobs ORDER BY id LIMIT 60"
        ).fetchall()


@app.post("/api/skills")
def skills(body: TextRequest):
    """Skills found in a piece of text (used to show what the resume contains)."""
    return extract_skills(body.text)


@app.get("/api/jobs/search")
def search(q: str, limit: int = 5):
    return tools.search_jobs(q, limit)


@app.get("/api/jobs/{job_id}")
def job(job_id: int):
    result = tools.get_job(job_id)
    if "error" in result:
        raise HTTPException(404, result["error"])
    return result


@app.post("/api/match")
def match(body: ResumeRequest):
    return tools.match_resume(body.resume_text)


@app.post("/api/skill-gap")
def gap(body: SkillGapRequest):
    result = tools.skill_gap(body.resume_text, body.job_id)
    if "error" in result:
        raise HTTPException(404, result["error"])
    return result


@app.post("/api/chat")
def chat(body: ChatRequest):
    """Run the LangGraph agent for one user message."""
    global _agent
    if not llm_configured():
        raise HTTPException(503, "No LLM key. Put OPENAI_API_KEY=... in backend/.env and restart the backend.")
    try:
        if _agent is None:
            _agent = build_agent()
        return run_agent(_agent, body.message, body.resume_text, body.history)
    except Exception as error:  # bad key, no credit, network down...
        raise HTTPException(502, f"The LLM call failed: {error}")

==================== backend/app/ingest.py ====================

"""
Load job postings into PostgreSQL: extract skills, compute the embedding, insert.

    python -m app.ingest              load data/sample_jobs.json (invented postings, works offline)
    python -m app.ingest --remotive   fetch real remote software jobs from the public Remotive API
"""
import json
import re
import sys
import urllib.request
from pathlib import Path

from app.db import connect, embed, init_db, to_pgvector
from app.skills import extract_skills

SAMPLE_FILE = Path(__file__).parent.parent / "data" / "sample_jobs.json"
REMOTIVE_URL = "https://remotive.com/api/remote-jobs?category=software-dev&limit=100"


def load_sample() -> list[dict]:
    return json.loads(SAMPLE_FILE.read_text())


def load_remotive() -> list[dict]:
    request = urllib.request.Request(REMOTIVE_URL, headers={"User-Agent": "InternInsider-demo"})
    with urllib.request.urlopen(request, timeout=30) as response:
        data = json.load(response)
    jobs = []
    for item in data["jobs"]:
        text = re.sub(r"<[^>]+>", " ", item["description"])  # strip HTML tags
        text = re.sub(r"\s+", " ", text).strip()[:3000]
        jobs.append({"title": item["title"], "company": item["company_name"],
                     "location": item.get("candidate_required_location", ""),
                     "category": item.get("category", ""), "url": item.get("url", ""),
                     "description": text})
    return jobs


def ingest(jobs: list[dict]) -> int:
    """Replace the jobs table with these postings. Returns how many were inserted."""
    init_db()
    with connect() as conn:
        conn.execute("TRUNCATE jobs RESTART IDENTITY")
        for job in jobs:
            text = job["title"] + ". " + job["description"]
            conn.execute(
                """
                INSERT INTO jobs (title, company, location, category, url, description, skills, embedding)
                VALUES (%s, %s, %s, %s, %s, %s, %s, %s::vector)
                """,
                (job["title"], job["company"], job.get("location", ""), job.get("category", ""),
                 job.get("url", ""), job["description"], extract_skills(text), to_pgvector(embed(text))),
            )
    return len(jobs)


if __name__ == "__main__":
    source = load_remotive if "--remotive" in sys.argv else load_sample
    print(f"Loaded {ingest(source())} job postings.")

class ScriptedLLM:
    """
    TEST ONLY. Stands in for the real LLM so the graph can be tested with no
    API key. It replays a fixed list of replies, one per call.
    """

    def __init__(self, replies):
        self.replies = list(replies)
        self.seen_tools = []

    def bind_tools(self, tools):
        self.seen_tools = [t.__name__ for t in tools]
        return self

    def invoke(self, messages):
        self.last_messages = messages
        return self.replies.pop(0)


def test_agent_loop_calls_tool_then_answers():
    llm = ScriptedLLM([
        # 1st LLM turn: "call search_jobs"
        AIMessage(content="", tool_calls=[{"name": "search_jobs", "args": {"query": "Swift iOS", "limit": 2}, "id": "call_1"}]),
        # 2nd LLM turn: a normal answer, which ends the loop
        AIMessage(content="Job 17, iOS Developer Intern at Trailmark, fits best."),
    ])
    agent = build_agent(llm)
    result = run_agent(agent, "Find me iOS jobs", resume_text=RESUME)

    assert llm.seen_tools == ["search_jobs", "get_job", "match_resume", "skill_gap", "interview_prep"]
    assert result["tool_calls"] == [{"name": "search_jobs", "args": {"query": "Swift iOS", "limit": 2}}]
    assert result["reply"].startswith("Job 17")
    # On its 2nd turn the LLM was shown the real tool output from PostgreSQL.
    tool_message = llm.last_messages[-1]
    assert isinstance(tool_message, ToolMessage)
    assert "iOS Developer Intern" in tool_message.content
    # The resume was placed in the system prompt.
    assert RESUME in llm.last_messages[0].content


def test_agent_answers_directly_when_no_tool_is_needed():
    agent = build_agent(ScriptedLLM([AIMessage(content="Hello! Paste your resume to start.")]))
    result = run_agent(agent, "hi")
    assert result == {"reply": "Hello! Paste your resume to start.", "tool_calls": []}


# --------------------------------------------------------------------- MCP

def test_mcp_server_exposes_the_five_tools():
    """Start the MCP server as a real subprocess and talk to it like an AI app would."""
    from mcp import ClientSession, StdioServerParameters
    from mcp.client.stdio import stdio_client

    async def scenario():
        params = StdioServerParameters(command=sys.executable, args=["-m", "app.mcp_server"])
        async with stdio_client(params) as (read, write):
            async with ClientSession(read, write) as session:
                await session.initialize()
                names = [t.name for t in (await session.list_tools()).tools]
                result = await session.call_tool("skill_gap", {"resume_text": RESUME, "job_id": 5})
                return names, result

    names, result = asyncio.run(scenario())
    assert names == ["search_jobs", "get_job", "match_resume", "skill_gap", "interview_prep"]
    assert not result.is_error
    assert '"match_percent"' in result.content[0].text and "57" in result.content[0].text

==================== frontend/src/api.ts ====================

// Everything the frontend knows about talking to the backend lives here.
const API_URL: string = import.meta.env.VITE_API_URL ?? "http://localhost:8001";

// These types mirror the JSON the FastAPI backend returns.
export type Job = {
  id: number;
  title: string;
  company: string;
  location: string;
  skills: string[];
  similarity?: number; // 0 to 1 from pgvector; only present on search results
};

export type JobDetail = Job & { description: string; category: string; url: string };

export type SkillGap = {
  job_id: number;
  have: string[];
  missing: string[];
  match_percent: number;
};

export type Health = { jobs: number; llm_configured: boolean };
export type ToolCall = { name: string; args: Record<string, unknown> };
export type ChatMessage = { role: "user" | "assistant"; content: string; tool_calls?: ToolCall[] };

// One helper for every request. Throws the server's error message on 4xx/5xx.
async function request<T>(path: string, body?: unknown): Promise<T> {
  const response = await fetch(API_URL + path, {
    method: body ? "POST" : "GET",
    headers: { "Content-Type": "application/json" },
    body: body ? JSON.stringify(body) : undefined,
  });
  const data = await response.json();
  if (!response.ok) throw new Error(typeof data.detail === "string" ? data.detail : "Request failed");
  return data;
}

export const getHealth = () => request<Health>("/api/health");
export const listJobs = () => request<Job[]>("/api/jobs");
export const getJob = (id: number) => request<JobDetail>(`/api/jobs/${id}`);
export const searchJobs = (query: string) =>
  request<Job[]>("/api/jobs/search?limit=8&q=" + encodeURIComponent(query));
export const matchResume = (resume_text: string) => request<Job[]>("/api/match", { resume_text });
export const extractSkills = (text: string) => request<string[]>("/api/skills", { text });
export const getSkillGap = (resume_text: string, job_id: number) =>
  request<SkillGap>("/api/skill-gap", { resume_text, job_id });

export const sendChat = (message: string, resume_text: string, history: ChatMessage[]) =>
  request<{ reply: string; tool_calls: ToolCall[] }>("/api/chat", {
    message,
    resume_text,
    // The backend is stateless, so earlier messages are sent along every time.
    history: history.map(({ role, content }) => ({ role, content })),
  });

==================== frontend/src/App.tsx ====================

import { useEffect, useRef, useState } from "react";
import {
  ChatMessage, Health, Job, JobDetail, SkillGap,
  extractSkills, getHealth, getJob, getSkillGap, listJobs, matchResume, searchJobs, sendChat,
} from "./api";

type Page = "jobs" | "agent" | "resume";

// DEMO ONLY: an invented resume so the app can be shown without typing one.
const SAMPLE_RESUME = `Computing Science student, third year.
Projects: a REST API in Python and FastAPI with PostgreSQL, packaged with Docker and tested with pytest.
A React and TypeScript dashboard. Coursework in SQL, statistics and machine learning with pandas.
Tools: Git, Linux.`;

const SUGGESTIONS = [
  "Which jobs fit my resume best?",
  "What am I missing for job 5?",
  "Give me interview questions for job 9",
  "Find internships that work with data",
];

export default function App() {
  const [page, setPage] = useState<Page>("jobs");
  const [health, setHealth] = useState<Health | null>(null);
  const [offline, setOffline] = useState(false);

  // The resume is kept in the browser (localStorage) so a refresh doesn't lose it.
  const [resume, setResume] = useState(() => localStorage.getItem("resume") ?? "");
  const [resumeSkills, setResumeSkills] = useState<string[]>([]);

  // Chat state lives here, not in the chat page, so it survives switching pages.
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [busy, setBusy] = useState(false);
  const [chatError, setChatError] = useState("");

  useEffect(() => {
    getHealth().then(setHealth).catch(() => setOffline(true));
  }, []);

  // Whenever the saved resume changes, ask the backend which skills it contains.
  useEffect(() => {
    localStorage.setItem("resume", resume);
    if (!resume.trim()) return setResumeSkills([]);
    extractSkills(resume).then(setResumeSkills).catch(() => setResumeSkills([]));
  }, [resume]);

  // Send one message to the LangGraph agent. Any page can call this.
  async function ask(text: string) {
    if (!text.trim() || busy) return;
    setPage("agent");
    const history = messages;
    const userMessage: ChatMessage = { role: "user", content: text };
    setMessages([...history, userMessage]);
    setBusy(true);
    setChatError("");
    try {
      const result = await sendChat(text, resume, history);
      setMessages([...history, userMessage, { role: "assistant", content: result.reply, tool_calls: result.tool_calls }]);
    } catch (e) {
      setChatError((e as Error).message);
    } finally {
      setBusy(false);
    }
  }

  const tabs: [Page, string][] = [["jobs", "Jobs"], ["agent", "AI copilot"], ["resume", "My resume"]];

  return (
    <div className="shell">
      <aside className="sidebar">
        <div className="brand">intern<span>insider</span></div>
        <nav>
          {tabs.map(([id, label]) => (
            <button key={id} className={page === id ? "nav active" : "nav"} onClick={() => setPage(id)}>
              {label}
              {id === "resume" && !resume.trim() && <span className="badge">add</span>}
            </button>
          ))}
        </nav>
        <div className="status">
          {offline && <div className="bad">Backend is not running</div>}
          {health && <div>{health.jobs} postings loaded</div>}
          {health && <div className={health.llm_configured ? "good" : "warn"}>AI copilot {health.llm_configured ? "ready" : "off (no API key)"}</div>}
          <div className="demo-note">DEMO: no login. Sample postings unless real ones were loaded.</div>
        </div>
      </aside>

      <main className="content">
        {page === "jobs" && <JobsPage resume={resume} resumeSkills={resumeSkills} onAsk={ask} goResume={() => setPage("resume")} />}
        {page === "agent" && <AgentPage messages={messages} busy={busy} error={chatError} onAsk={ask} hasResume={!!resume.trim()} />}
        {page === "resume" && <ResumePage resume={resume} resumeSkills={resumeSkills} onSave={setResume} />}
      </main>
    </div>
  );
}

// ------------------------------------------------------------------ Jobs page

function JobsPage(props: { resume: string; resumeSkills: string[]; onAsk: (text: string) => void; goResume: () => void }) {
  const { resume, resumeSkills, onAsk, goResume } = props;
  const [query, setQuery] = useState("");
  const [jobs, setJobs] = useState<Job[]>([]);
  const [heading, setHeading] = useState("All postings");
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [openId, setOpenId] = useState<number | null>(null);

  // Run one of the three ways of loading jobs and show the result.
  async function load(title: string, fetchJobs: () => Promise<Job[]>) {
    setLoading(true);
    setError("");
    try {
      setJobs(await fetchJobs());
      setHeading(title);
    } catch (e) {
      setError((e as Error).message === "Failed to fetch" ? "Can't reach the backend. Is it running?" : (e as Error).message);
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => { load("All postings", listJobs); }, []);

  return (
    <>
      <h1>Find a role that fits</h1>
      <p className="lead">Search by what you want to do, in your own words. It matches on meaning, so the posting doesn't need to contain your exact words.</p>

      <form className="searchbar" onSubmit={(e) => { e.preventDefault(); load(`Closest to "${query}"`, () => searchJobs(query)); }}>
        <input value={query} onChange={(e) => setQuery(e.target.value)} placeholder='e.g. "making apps for phones" or "working with data"' required />
        <button className="primary" type="submit">Search</button>
        <button type="button" disabled={!resume.trim()} title={resume.trim() ? "" : "Add your resume first"}
          onClick={() => load("Best matches for your resume", () => matchResume(resume))}>
          Match my resume
        </button>
        <button type="button" className="link" onClick={() => { setQuery(""); load("All postings", listJobs); }}>Show all</button>
      </form>

      {!resume.trim() && (
        <p className="hint">Add your resume to see which skills you already have for each job. <button className="link" onClick={goResume}>Add resume</button></p>
      )}
      {error && <p className="error">{error}</p>}

      <h2>{heading} <span className="count">{loading ? "loading…" : jobs.length}</span></h2>
      <div className="grid">
        {jobs.map((job) => {
          const have = job.skills.filter((s) => resumeSkills.includes(s));
          return (
            <button className="card" key={job.id} data-testid="job" onClick={() => setOpenId(job.id)}>
              <div className="card-top">
                <span className="job-id">#{job.id}</span>
                {job.similarity !== undefined && <span className="sim" title="Cosine similarity from pgvector">similarity {job.similarity.toFixed(2)}</span>}
              </div>
              <h3>{job.title}</h3>
              <div className="company">{job.company} · {job.location}</div>
              <Chips skills={job.skills} have={resumeSkills} />
              {resumeSkills.length > 0 && job.skills.length > 0 && (
                <Meter percent={Math.round((100 * have.length) / job.skills.length)} label={`${have.length} of ${job.skills.length} skills`} />
              )}
            </button>
          );
        })}
      </div>
      {!loading && jobs.length === 0 && !error && <p className="hint">No postings. Run the backend start script to load them.</p>}

      {openId !== null && <JobDrawer id={openId} resume={resume} onClose={() => setOpenId(null)} onAsk={onAsk} />}
    </>
  );
}

// Slide-in panel with one job's details and the skill gap.
function JobDrawer({ id, resume, onClose, onAsk }: { id: number; resume: string; onClose: () => void; onAsk: (text: string) => void }) {
  const [job, setJob] = useState<JobDetail | null>(null);
  const [gap, setGap] = useState<SkillGap | null>(null);

  useEffect(() => {
    getJob(id).then(setJob).catch(() => setJob(null));
    if (resume.trim()) getSkillGap(resume, id).then(setGap).catch(() => setGap(null));
  }, [id, resume]);

  return (
    <div className="overlay" onMouseDown={(e) => { if (e.target === e.currentTarget) onClose(); }}>
      <section className="drawer" role="dialog" aria-label="Job details" data-testid="drawer">
        <button className="close" onClick={onClose} aria-label="Close">×</button>
        {!job ? <p className="hint">Loading…</p> : (
          <>
            <div className="job-id">Posting #{job.id}</div>
            <h2>{job.title}</h2>
            <div className="company">{job.company} · {job.location}</div>
            <p className="description">{job.description}</p>

            <h4>Skill gap</h4>
            {!resume.trim() && <p className="hint">Add your resume to see what you have and what's missing.</p>}
            {gap && (
              <div data-testid="gap">
                <Meter percent={gap.match_percent} label={`${gap.match_percent}% of listed skills`} />
                <div className="gap-row"><span className="gap-label">You have</span><Chips skills={gap.have} have={gap.have} empty="none yet" /></div>
                <div className="gap-row"><span className="gap-label">To learn</span><Chips skills={gap.missing} have={[]} missing empty="nothing, you cover them all" /></div>
                <p className="fine">This is skill overlap from a word list, not a prediction of getting hired.</p>
              </div>
            )}

            <h4>Ask the AI copilot about this job</h4>
            <div className="actions">
              <button onClick={() => { onClose(); onAsk(`Give me interview questions for job ${job.id}`); }}>Interview questions</button>
              <button onClick={() => { onClose(); onAsk(`What am I missing for job ${job.id}, and what should I learn first?`); }}>What should I learn first?</button>
            </div>
          </>
        )}
      </section>
    </div>
  );
}

// ------------------------------------------------------------------ Chat page

function AgentPage(props: { messages: ChatMessage[]; busy: boolean; error: string; onAsk: (text: string) => void; hasResume: boolean }) {
  const { messages, busy, error, onAsk, hasResume } = props;
  const [input, setInput] = useState("");
  const bottom = useRef<HTMLDivElement>(null);

  // Scroll to the newest message whenever the list changes.
  useEffect(() => { bottom.current?.scrollIntoView({ behavior: "smooth" }); }, [messages, busy]);

  return (
    <>
      <h1>AI copilot</h1>
      <p className="lead">A LangGraph agent with five tools. It decides which tool to call, and the grey line above each answer shows what it used.</p>
      {!hasResume && <p className="hint">No resume added yet. Resume questions will work better once you add one.</p>}

      <div className="chat">
        {messages.length === 0 && (
          <div className="empty-chat">
            <p>Try one of these:</p>
            {SUGGESTIONS.map((s) => <button key={s} className="suggestion" onClick={() => onAsk(s)}>{s}</button>)}
          </div>
        )}
        {messages.map((m, i) => (
          <div key={i} className={"bubble-row " + m.role}>
            <div>
              {m.tool_calls && m.tool_calls.length > 0 && (
                <div className="tools" data-testid="tools">
                  {m.tool_calls.map((t, j) => <span key={j} className="tool" title={JSON.stringify(t.args)}>used {t.name}</span>)}
                </div>
              )}
              <div className="bubble">{m.content}</div>
            </div>
          </div>
        ))}
        {busy && <div className="bubble-row assistant"><div className="bubble typing"><i /><i /><i /></div></div>}
        {error && <p className="error" data-testid="chat-error">{error}</p>}
        <div ref={bottom} />
      </div>

      <form className="composer" onSubmit={(e) => { e.preventDefault(); onAsk(input); setInput(""); }}>
        <input value={input} onChange={(e) => setInput(e.target.value)} placeholder="Ask about jobs, skill gaps or interview prep" disabled={busy} required />
        <button className="primary" type="submit" disabled={busy}>Send</button>
      </form>
    </>
  );
}

// ---------------------------------------------------------------- Resume page

function ResumePage({ resume, resumeSkills, onSave }: { resume: string; resumeSkills: string[]; onSave: (text: string) => void }) {
  const [draft, setDraft] = useState(resume);
  const saved = draft === resume;

  return (
    <>
      <h1>My resume</h1>
      <p className="lead">Paste your resume as plain text. It stays in this browser and is sent to the backend only when you search, match or chat.</p>
      <textarea rows={12} value={draft} onChange={(e) => setDraft(e.target.value)} placeholder="Paste your resume here" />
      <div className="actions">
        <button className="primary" disabled={saved} onClick={() => onSave(draft)}>{saved && resume ? "Saved" : "Save resume"}</button>
        <button onClick={() => { setDraft(SAMPLE_RESUME); onSave(SAMPLE_RESUME); }}>Use sample resume</button>
        <span className="fine">DEMO: the sample resume is invented.</span>
      </div>

      <h2>Skills found in your resume <span className="count">{resumeSkills.length}</span></h2>
      <Chips skills={resumeSkills} have={resumeSkills} empty="Save a resume to see the skills it mentions." />
      <p className="fine">Found by matching against a fixed list of skills, so a skill written in an unusual way can be missed.</p>
    </>
  );
}

// ------------------------------------------------------------ small pieces

function Chips({ skills, have, missing, empty }: { skills: string[]; have: string[]; missing?: boolean; empty?: string }) {
  if (skills.length === 0) return <span className="fine">{empty ?? ""}</span>;
  return (
    <div className="chips">
      {skills.map((s) => (
        <span key={s} className={"chip" + (have.includes(s) ? " have" : missing ? " missing" : "")}>{s}</span>
      ))}
    </div>
  );
}

function Meter({ percent, label }: { percent: number; label: string }) {
  return (
    <div className="meter" aria-label={label}>
      <div className="meter-track"><div className="meter-fill" style={{ width: percent + "%" }} /></div>
      <span>{label}</span>
    </div>
  );
}
